# --- B3: acceso seguro (EVID-014 a EVID-026) ---

  local-user eicladmin password            # contrasena interactiva <REDACTADO>

  quit

 ssh user eicladmin service-type stelnet   # cambiado a 'all' en B9

  rule 5 permit source <IP-PC-ADMIN> 0     # PC de administracion (IP redactada)
  rule 10 permit source 10.10.10.0 0.0.0.255
  rule 100 deny
  quit

  protocol inbound ssh                     # valor por defecto, no aparece en la configuracion

  idle-timeout 10 0                        # valor por defecto, no aparece en la configuracion

  acl 2000 inbound
  quit

  authentication-mode aaa                  # pide verificar la contrasena de consola anterior
  idle-timeout 10 0
  return
